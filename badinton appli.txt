    <script>
        const state = {
            mode: 'time',
            limit: 5,
            scoreA: 0,
            scoreB: 0,
            history: [],
            configA: { zone: 'back', size: 22 },
            configB: { zone: 'back', size: 22 },
            timeLeft: 0,
            timerInterval: null,
            isPaused: false,
            matchEnded: false
        };

        function drawCourts() {
            const containers = document.querySelectorAll('.lines-container');
            containers.forEach(container => {
                container.innerHTML = '';
                const hLines = [{ t: 0, h: 2 }, { t: 6, h: 1 }, { t: 94, h: 1 }, { t: 100, h: 2 }];
                const vLines = [{ l: 0, w: 2 }, { l: 5, w: 1 }, { l: 35, w: 1 }, { l: 65, w: 1 }, { l: 95, w: 1 }, { l: 100, w: 2 }];
                hLines.forEach(line => {
                    const el = document.createElement('div'); el.className = 'line w-full';
                    el.style.top = `calc(${line.t}% - ${line.h/2}px)`; el.style.height = `${line.h}px`;
                    container.appendChild(el);
                });
                vLines.forEach(line => {
                    const el = document.createElement('div'); el.className = 'line h-full';
                    el.style.left = `calc(${line.l}% - ${line.w/2}px)`; el.style.width = `${line.w}px`;
                    container.appendChild(el);
                });
                const leftMedian = document.createElement('div');
                leftMedian.className = 'line'; leftMedian.style.top = '50%'; leftMedian.style.left = '0'; leftMedian.style.width = '35%'; leftMedian.style.height = '1px';
                container.appendChild(leftMedian);
                const rightMedian = document.createElement('div');
                rightMedian.className = 'line'; rightMedian.style.top = '50%'; rightMedian.style.right = '0'; rightMedian.style.width = '35%'; rightMedian.style.height = '1px';
                container.appendChild(rightMedian);
            });
        }

        function setMode(m) {
            state.mode = m;
            const range = document.getElementById('limitRange');
            const label = document.getElementById('limitLabel');
            if(m === 'time') {
                range.min=1; range.max=10; range.value=5; label.innerText="Minutes (1-10)";
                document.getElementById('btnModeTime').classList.replace('border-slate-700', 'border-yellow-500');
                document.getElementById('btnModePoints').classList.replace('border-yellow-500', 'border-slate-700');
            } else {
                range.min=1; range.max=21; range.value=11; label.innerText="Points (1-21)";
                document.getElementById('btnModePoints').classList.replace('border-slate-700', 'border-yellow-500');
                document.getElementById('btnModeTime').classList.replace('border-yellow-500', 'border-slate-700');
            }
            updateLimitDisplay();
        }

        function updateLimitDisplay() {
            state.limit = parseInt(document.getElementById('limitRange').value);
            document.getElementById('limitDisplay').innerText = state.limit;
        }
        document.getElementById('limitRange').oninput = updateLimitDisplay;

        function goToTactics() {
            document.getElementById('setupView').classList.add('hidden-view');
            document.getElementById('tacticView').classList.remove('hidden-view');
            drawCourts();
            updateSetupOverlays();
        }

        function setDiff(player, size, btn) {
            if(player === 'A') state.configA.size = size;
            else state.configB.size = size;
            const btns = btn.parentNode.querySelectorAll('button');
            btns.forEach(b => b.classList.remove('active'));
            btn.classList.add('active');
            updateSetupOverlays();
        }

        function updateSetupOverlays() {
            state.configA.zone = document.getElementById('setupZoneA').value;
            state.configB.zone = document.getElementById('setupZoneB').value;
            updateOverlay(document.getElementById('setupOverlayA'), state.configA, 'right');
            updateOverlay(document.getElementById('setupOverlayB'), state.configB, 'left');
        }

        function startMatch() {
            document.getElementById('tacticView').classList.add('hidden-view');
            document.getElementById('matchView').classList.remove('hidden-view');
            drawCourts();
            if(state.mode === 'time') {
                state.timeLeft = state.limit * 60;
                document.getElementById('gameLimitInfo').innerText = `CHRONO: ${state.limit}m`;
                resumeTimer();
            } else {
                document.getElementById('gameLimitInfo').innerText = `CIBLE: ${state.limit} pts`;
                document.getElementById('gameTimer').innerText = "POINTS";
                document.getElementById('pauseBtn').classList.add('hidden');
            }
            updateUI();
        }

        function toggleTimer() { state.isPaused ? resumeTimer() : pauseTimer(); }
        function pauseTimer() {
            clearInterval(state.timerInterval); state.isPaused = true;
            document.getElementById('pauseIcon').innerHTML = '<path d="M8 5v14l11-7z"/>';
        }
        function resumeTimer() {
            state.isPaused = false;
            document.getElementById('pauseIcon').innerHTML = '<path d="M6 19h4V5H6v14zm8-14v14h4V5h-4z"/>';
            state.timerInterval = setInterval(() => {
                state.timeLeft--;
                const m = Math.floor(state.timeLeft / 60); const s = state.timeLeft % 60;
                document.getElementById('gameTimer').innerText = `${m.toString().padStart(2,'0')}:${s.toString().padStart(2,'0')}`;
                if(state.timeLeft <= 0) endGame();
            }, 1000);
        }

        const court = document.getElementById('court');
        court.addEventListener('click', (e) => {
            if(state.matchEnded || (state.mode === 'time' && state.isPaused)) return;
            const rect = court.getBoundingClientRect();
            const xP = ((e.clientX - rect.left) / rect.width) * 100;
            const yP = ((e.clientY - rect.top) / rect.height) * 100;
            const scoringPlayer = xP > 50 ? 'A' : 'B';
            let isBonus = false;
            const cfg = scoringPlayer === 'A' ? state.configA : state.configB;
            const relativeX = scoringPlayer === 'A' ? xP : (100 - xP);
            if(cfg.zone === 'back' && relativeX > (100 - cfg.size)) isBonus = true;
            if(cfg.zone === 'front' && relativeX >= 50 && relativeX < (50 + cfg.size)) isBonus = true;
            if(cfg.zone === 'top' && yP < cfg.size) isBonus = true;
            if(cfg.zone === 'bottom' && yP > (100 - cfg.size)) isBonus = true;
            addPoint(scoringPlayer, isBonus ? 3 : 1, xP, yP, false);
        });

        function handleFault(fBy) {
            if(state.matchEnded || (state.mode === 'time' && state.isPaused)) return;
            const scoring = fBy === 'A' ? 'B' : 'A';
            addPoint(scoring, 1, scoring==='A'?75:25, 50, true);
        }

        function addPoint(player, pts, x, y, fault) {
            player === 'A' ? state.scoreA += pts : state.scoreB += pts;
            const id = Date.now();
            state.history.push({ id, player, points: pts, x, y, fault });
            renderImpact({ id, player, points: pts, x, y, fault });
            updateUI();
            if(state.mode==='points' && (state.scoreA>=state.limit || state.scoreB>=state.limit)) endGame();
        }

        function updateUI() {
            document.getElementById('scoreA').innerText = state.scoreA.toString().padStart(2,'0');
            document.getElementById('scoreB').innerText = state.scoreB.toString().padStart(2,'0');
            updateOverlay(document.getElementById('overlayA'), state.configA, 'right');
            updateOverlay(document.getElementById('overlayB'), state.configB, 'left');
        }

        function updateOverlay(el, cfg, side) {
            el.style = `position: absolute; pointer-events: none; z-index: 10; background: ${side==='right'?'rgba(239,68,68,0.2)':'rgba(249,115,22,0.2)'}; border: 1px solid ${side==='right'?'rgba(239,68,68,0.4)':'rgba(249,115,22,0.4)'};`;
            el.style.top = '0'; el.style.bottom = '0'; el.style.height = '100%'; el.style.width = '50%';
            if(side === 'right') el.style.left = '50%'; else el.style.left = '0';
            if(cfg.zone === 'back') {
                el.style.width = cfg.size + '%';
                if(side === 'right') el.style.left = (100 - cfg.size) + '%';
            } else if(cfg.zone === 'front') {
                el.style.width = cfg.size + '%';
                if(side === 'left') el.style.left = (50 - cfg.size) + '%';
            } else if(cfg.zone === 'top') {
                el.style.height = cfg.size + '%';
            } else if(cfg.zone === 'bottom') {
                el.style.height = cfg.size + '%'; el.style.top = 'auto'; el.style.bottom = '0';
            }
        }

        function renderImpact(h) {
            const d = document.createElement('div');
            d.id = `impact-${h.id}`;
            d.className = `absolute w-6 h-6 rounded-full flex items-center justify-center text-[10px] font-black impact-point z-40 border-2 border-white/50`;
            d.style.left = `calc(${h.x}% - 12px)`; d.style.top = `calc(${h.y}% - 12px)`;
            d.style.backgroundColor = h.player === 'A' ? '#ef4444' : '#f97316';
            if(h.points < 3) { d.style.backgroundColor = 'transparent'; d.style.borderColor = h.player==='A'?'#ef4444':'#f97316'; d.style.color = d.style.borderColor; }
            d.innerText = h.points === 3 ? '★' : '1';
            document.getElementById('impactsContainer').appendChild(d);
        }

        function undoLast() {
            const h = state.history.pop(); if(!h) return;
            h.player==='A'?state.scoreA-=h.points:state.scoreB-=h.points;
            const el = document.getElementById(`impact-${h.id}`); if(el) el.remove();
            updateUI();
        }

        function stopMatchEarly() { if(confirm("Finir le match ?")) endGame(); }

        function endGame() {
            state.matchEnded = true; clearInterval(state.timerInterval);
            document.getElementById('matchView').classList.add('hidden-view');
            document.getElementById('statsView').classList.remove('hidden-view');
            document.getElementById('finalScoreA').innerText = state.scoreA;
            document.getElementById('finalScoreB').innerText = state.scoreB;
            const a_bonus = state.history.filter(h => h.player === 'A' && h.points === 3).length;
            const a_norm = state.history.filter(h => h.player === 'A' && h.points === 1).length;
            const b_bonus = state.history.filter(h => h.player === 'B' && h.points === 3).length;
            const b_norm = state.history.filter(h => h.player === 'B' && h.points === 1).length;
            document.getElementById('bonusCountA').innerText = a_bonus;
            document.getElementById('bonusCountB').innerText = b_bonus;
            createChart('chartA', [a_norm, a_bonus], '#ef4444');
            createChart('chartB', [b_norm, b_bonus], '#f97316');
        }

        function createChart(id, d, color) {
            new Chart(document.getElementById(id), {
                type: 'doughnut',
                data: { 
                    labels: ['Point classique', 'Point cible'], 
                    datasets: [{ 
                        data: d, 
                        backgroundColor: [color+'22', color], 
                        borderColor: color, 
                        borderWidth: 2 
                    }] 
                },
                options: { 
                    plugins: { 
                        legend: { 
                            display: true, 
                            position: 'bottom',
                            labels: {
                                color: '#94a3b8',
                                font: { size: 10, weight: 'bold', family: 'Inter' },
                                padding: 15,
                                usePointStyle: true,
                                pointStyle: 'circle'
                            }
                        } 
                    }, 
                    cutout: '65%' 
                }
            });
        }
